// ===== BUAT USER ID DENGAN ADD-ON ===== async function buatUserID(orderId, paket, key) { // ===== HANYA TANYAKAN ADD-ON JIKA PAKET BASIC ===== let addonPrint = false; if (paket === 'basic') { const tanyaAddon = confirm( 'Apakah user ini membeli Add-On Print (Rp 100.000)?\n\n' + 'Klik OK jika Ya, Cancel jika Tidak' ); addonPrint = tanyaAddon ? true : false; } else { // Premium & Pro Max otomatis punya akses print addonPrint = true; } // ===== 2. GENERATE USER ID UNIK ===== const userId = generateUniqueUserID(); const kodeUnik = generateUniqueKode(); try { // ===== 3. SIMPAN KE FIREBASE ===== await db.ref('transactions/' + key + '/user_id').set(userId); await db.ref('transactions/' + key + '/verified').set(true); // ===== 4. SIMPAN KE SUPABASE ===== const { data: existingUser } = await sb.from('users').select('*').eq('id', userId).single(); if (!existingUser) { const { data, error } = await sb.from('users').insert({ id: userId, kode_unik: kodeUnik, paket: paket, role: 'management', status: 'Aktif', addon_print: addonPrint, created_at: new Date().toISOString() }).select().single(); if (error) throw error; console.log("User berhasil dibuat di Supabase:", data); } // ===== 5. TAMPILKAN HASIL ===== document.getElementById('hasilKode').textContent = kodeUnik; document.getElementById('hasilUserId').textContent = userId; document.getElementById('hasilAddon').textContent = addonPrint ? '✅ Ya (Rp 100.000)' : '❌ Tidak'; document.getElementById('hasilVerifikasi').classList.remove('hide'); sessionStorage.setItem('lastGeneratedUserId', userId); sessionStorage.setItem('lastGeneratedKode', kodeUnik); await loadUsers(); renderTransactions(); } catch (err) { alert('❌ Gagal membuat User ID: ' + err.message); } }
